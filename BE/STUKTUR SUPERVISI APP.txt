use rswj_app;
-- =====================================================
-- MASTER SUPERVISI
-- =====================================================

CREATE TABLE supervisi (
    supervisi_id VARCHAR(22) NOT NULL,
    tanggal_supervisi DATE NOT NULL,
    periode_awal DATETIME NOT NULL,
    periode_akhir DATETIME NOT NULL,
    status ENUM('OPEN','REVIEW','FINAL')
        NOT NULL DEFAULT 'OPEN',
    created_by VARCHAR(50),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_by VARCHAR(50),
    updated_at DATETIME NULL,
    is_active TINYINT(1) DEFAULT 1,
    PRIMARY KEY (supervisi_id),
    UNIQUE KEY uk_supervisi_tanggal (
        tanggal_supervisi
    )
) ENGINE=InnoDB DEFAULT CHARSET=utf8;


-- =====================================================
-- HD
-- =====================================================

CREATE TABLE supervisi_hd (
    supervisi_hd_id VARCHAR(22) NOT NULL,
    supervisi_id VARCHAR(22) NOT NULL,
    jumlah_pasien INT DEFAULT 0,
    pasien_reguler INT DEFAULT 0,
    pasien_isolasi INT DEFAULT 0,
    capd INT DEFAULT 0,
    pembiayaan_bpjs INT DEFAULT 0,
    jumlah_tindakan INT DEFAULT 0,
    catatan TEXT,
    created_by VARCHAR(50),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_by VARCHAR(50),
    updated_at DATETIME NULL,
    is_active TINYINT(1) DEFAULT 1,
    PRIMARY KEY (supervisi_hd_id),
    UNIQUE KEY uk_supervisi_hd (
        supervisi_id
    ),
    CONSTRAINT fk_supervisi_hd
        FOREIGN KEY (supervisi_id)
        REFERENCES supervisi(supervisi_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- =====================================================
-- DATA MUTU & KESELAMATAN PASIEN

-- =====================================================
-- FOKUS PERHATIAN DIREKSI
-- =====================================================

CREATE TABLE direksi_issue (
    direksi_issue_id VARCHAR(22) NOT NULL,
    judul VARCHAR(255) NOT NULL,
    deskripsi TEXT,
    prioritas ENUM(
        'RENDAH',
        'SEDANG',
        'TINGGI',
        'KRITIS'
    ) DEFAULT 'SEDANG',
    status ENUM(
        'OPEN',
        'PROGRESS',
        'DONE',
        'CANCEL'
    ) DEFAULT 'OPEN',
    is_fokus_hari_ini TINYINT(1) DEFAULT 0,
    pic VARCHAR(100),
    tanggal_mulai DATE,
    target_selesai DATE,
    tanggal_selesai DATE NULL,
    created_by VARCHAR(50),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_by VARCHAR(50),
    updated_at DATETIME NULL,
    is_active TINYINT(1) DEFAULT 1,
    PRIMARY KEY (direksi_issue_id),
    KEY idx_status (status),
    KEY idx_fokus (is_fokus_hari_ini)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;


-- =====================================================
-- RENCANA AKSI CEPAT
-- =====================================================

CREATE TABLE direksi_plan (
    direksi_plan_id VARCHAR(22) NOT NULL,
    direksi_issue_id VARCHAR(22) NOT NULL,
    uraian_tindakan TEXT NOT NULL,
    pic VARCHAR(100),
    status ENUM(
        'OPEN',
        'PROGRESS',
        'DONE',
        'CANCEL'
    ) DEFAULT 'OPEN',
    target_selesai DATE,
    tanggal_selesai DATE NULL,
    created_by VARCHAR(50),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_by VARCHAR(50),
    updated_at DATETIME NULL,
    is_active TINYINT(1) DEFAULT 1,
    PRIMARY KEY (direksi_plan_id),
    KEY idx_direksi_plan_issue (
        direksi_issue_id
    ),
    CONSTRAINT fk_direksi_plan
        FOREIGN KEY (direksi_issue_id)
        REFERENCES direksi_issue(direksi_issue_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
